== Get the RADIUS service address ==

{{- if eq .Values.freeradius.service.type "LoadBalancer" }}
  NOTE: It may take a few minutes for the LoadBalancer IP to be available.
  kubectl get svc {{ include "freeradius.fullname" . }} -n {{ .Release.Namespace }} -w
{{- else if eq .Values.freeradius.service.type "NodePort" }}
  export NODE_IP=$(kubectl get nodes -o jsonpath="{.items[0].status.addresses[0].address}")
  echo "RADIUS Auth: $NODE_IP:$(kubectl get svc {{ include "freeradius.fullname" . }} -n {{ .Release.Namespace }} -o jsonpath='{.spec.ports[0].nodePort}')"
{{- else }}
  kubectl port-forward svc/{{ include "freeradius.fullname" . }} 1812:1812/udp -n {{ .Release.Namespace }}
{{- end }}

== Test Authentication ==

  # Add test user
  kubectl exec -n {{ .Release.Namespace }} {{ include "freeradius.fullname" . }}-mysql-0 -- \
    mysql -u {{ include "freeradius.mysql.user" . }} -p'<PASSWORD>' {{ include "freeradius.mysql.database" . }} -e \
    "INSERT INTO radcheck (username, attribute, op, value) VALUES ('testuser', 'Cleartext-Password', ':=', 'testpass') ON DUPLICATE KEY UPDATE value='testpass';"

  # Test RADIUS auth (use the default localhost client secret from clients.conf)
  kubectl exec -n {{ .Release.Namespace }} deploy/{{ include "freeradius.fullname" . }} -- \
    radtest testuser testpass 127.0.0.1 0 testing123
  # NOTE: 'testing123' is the default FreeRADIUS localhost client secret.
  # To test from outside the pod, use the RADIUS_SECRET you configured.
